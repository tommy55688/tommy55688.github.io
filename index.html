        const App = () => {

            const [viewMode, setViewMode] = useState('mobile'); 

            const [activeTab, setActiveTab] = useState('eco_linepay'); 

            

            const [aiAnalysis, setAiAnalysis] = useState("");

            const [isAiLoading, setIsAiLoading] = useState(false);

            const [showAiModal, setShowAiModal] = useState(false);


            const fileInputRef = useRef(null);


            const today = new Date();

            const [filterYear, setFilterYear] = useState(today.getFullYear());

            const [filterMonth, setFilterMonth] = useState(today.getMonth());


            const [transactions, setTransactions] = useState(() => {

                const saved = localStorage.getItem('dbs_transactions_v7');

                if (saved) return JSON.parse(saved);

                

                const savedV6 = localStorage.getItem('dbs_transactions_v6');

                if (savedV6) {

                    const parsed = JSON.parse(savedV6);

                    return parsed.map(t => ({ ...t, card: t.card === 'eco' ? 'eco_linepay' : t.card }));

                }

                return [];

            });


            const [form, setForm] = useState({

                date: new Date().toISOString().split('T')[0],

                card: 'eco_linepay',

                name: '',

                amount: ''

            });


            const [editingId, setEditingId] = useState(null);

            const [editForm, setEditForm] = useState({});


            useEffect(() => {

                localStorage.setItem('dbs_transactions_v7', JSON.stringify(transactions));

            }, [transactions]);


            const CARD_CONFIG = {

                eco_linepay: { name: 'eco 卡 (Line Pay 9%)', shortName: 'eco (Line Pay)', baseRate: 0.01, bonusRate: 0.09, spendCap: 11111, bonusPointCap: 1000 },

                eco_overseas: { name: 'eco 卡 (海外 4%)', shortName: 'eco (海外)', baseRate: 0.01, bonusRate: 0.04, spendCap: 15000, bonusPointCap: 600 },

                legend: { name: '傳說對決卡', shortName: '傳說卡', baseRate: 0.002, bonusRate: 0.098, spendCap: 11363, bonusPointCap: 1000 }

            };


            const getCardTheme = (cardKey) => {

                switch(cardKey) {

                    case 'legend': return { text: 'text-blue-700', badge: 'bg-blue-100 text-blue-700', border: 'border-l-4 border-l-blue-500' };

                    case 'eco_overseas': return { text: 'text-purple-700', badge: 'bg-purple-100 text-purple-700', border: 'border-l-4 border-l-purple-500' };

                    default: return { text: 'text-green-700', badge: 'bg-green-100 text-green-700', border: 'border-l-4 border-l-green-500' };

                }

            };


            const formatDateLocal = (date) => {

                const y = date.getFullYear();

                const m = String(date.getMonth() + 1).padStart(2, '0');

                const d = String(date.getDate()).padStart(2, '0');

                return `${y}-${m}-${d}`;

            };


            const callGeminiAPI = async (prompt) => {

                const apiKey = "";

                if (!apiKey || !navigator.onLine) {

                    return "目前處於離線模式或未設定 API 金鑰，無法提供 AI 分析建議。您可以繼續使用本機的記帳與計算功能！";

                }

                const url = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-09-2025:generateContent?key=${apiKey}`;

                

                const payload = {

                    contents: [{ parts: [{ text: prompt }] }],

                    systemInstruction: { parts: [{ text: "你是一位專業的星展信用卡理財助手。請分析使用者的消費數據，給予簡短、專業且具備繁體中文口吻的理財建議。" }] }

                };


                for (let i = 0; i < 3; i++) {

                    try {

                        const response = await fetch(url, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(payload) });

                        if (!response.ok) throw new Error('API Error');

                        const data = await response.json();

                        return data.candidates?.[0]?.content?.parts?.[0]?.text;

                    } catch (error) {

                        if (i === 2) throw error;

                        await new Promise(r => setTimeout(r, 1000));

                    }

                }

            };


            const processedData = useMemo(() => {

                const sorted = [...transactions].sort((a, b) => new Date(a.date) - new Date(b.date));

                const monthlyUsage = {}; 


                return sorted.map((t) => {

                    const dateObj = new Date(t.date);

                    const day = dateObj.getDate();

                    const amount = parseFloat(t.amount) || 0;

                    const configKey = CARD_CONFIG[t.card] ? t.card : 'eco_linepay'; 

                    const config = CARD_CONFIG[configKey];


                    const consumeMonthKey = `${dateObj.getFullYear()}-${dateObj.getMonth()}_${configKey}`;

                    const prevSpend = monthlyUsage[consumeMonthKey] || 0;

                    

                    const eligibleForBonus = Math.max(0, Math.min(amount, config.spendCap - prevSpend));

                    const exceeded = Math.max(0, amount - eligibleForBonus);

                    monthlyUsage[consumeMonthKey] = prevSpend + amount;


                    let billYear = dateObj.getFullYear();

                    let billMonth = dateObj.getMonth();

                    if (day >= 12) {

                        billMonth += 1;

                        if (billMonth > 11) { billMonth = 0; billYear += 1; }

                    }

                    const billingCycleStr = `${billYear}/${String(billMonth + 1).padStart(2, '0')} 帳單`;


                    const rewardBase = Math.round(amount * config.baseRate);

                    const rewardBonus = Math.round(eligibleForBonus * config.bonusRate);

                    const creditDateBonusObj = new Date(dateObj.getFullYear(), dateObj.getMonth() + 2, 1);

                    

                    return {

                        ...t, card: configKey, eligibleForBonus, exceeded, rewardBase, rewardBonus, totalReward: rewardBase + rewardBonus,

                        billingCycle: billingCycleStr, creditDateBonus: formatDateLocal(creditDateBonusObj),

                        baseRateLabel: (config.baseRate * 100).toFixed(1) + '%', bonusRateLabel: (config.bonusRate * 100).toFixed(1) + '%',

                        currentCap: config.spendCap

                    };

                });

            }, [transactions]);


            const summary = useMemo(() => {

                const config = CARD_CONFIG[activeTab];

                const dataInMonth = processedData.filter(t => {

                    const d = new Date(t.date);

                    return d.getFullYear() === filterYear && d.getMonth() === filterMonth && t.card === activeTab;

                });


                const totalSpent = dataInMonth.reduce((sum, t) => sum + (parseFloat(t.amount) || 0), 0);

                const totalBonusPoints = dataInMonth.reduce((sum, t) => sum + t.rewardBonus, 0);

                const remainingQuota = Math.max(0, config.spendCap - totalSpent);

                const isReached = totalSpent >= config.spendCap;

                const isPointsMax = totalBonusPoints >= config.bonusPointCap;


                return { totalSpent, totalBonusPoints, remainingQuota, isReached, isPointsMax, percent: Math.min(100, (totalSpent / config.spendCap) * 100), dataInMonth, currentCap: config.spendCap, bonusPointCap: config.bonusPointCap };

            }, [processedData, activeTab, filterMonth, filterYear]);


            const handleAiAnalysis = async () => {

                setIsAiLoading(true);

                setShowAiModal(true);

                try {

                    const spendDesc = summary.dataInMonth.map(t => `${t.name}($${t.amount})`).join(', ');

                    const prompt = `當前分析卡別：${CARD_CONFIG[activeTab].name}\n查詢月份：${filterYear}年${filterMonth + 1}月\n目前已消費：$${summary.totalSpent}\n本卡別加碼上限：$${summary.currentCap}\n剩餘加碼額度：$${summary.remainingQuota}\n目前已領加碼點數：${summary.totalBonusPoints} 點\n本月消費明細：${spendDesc || '無'}\n請根據以上資訊，給予這張卡片 3 點具體的理財策略建議。`;

                    const result = await callGeminiAPI(prompt);

                    setAiAnalysis(result);

                } catch (err) {

                    setAiAnalysis("AI 分析暫時無法使用，請確認網路連線。");

                } finally {

                    setIsAiLoading(false);

                }

            };


            const addTransaction = (e) => {

                e.preventDefault();

                if (!form.amount || !form.name) return;

                setTransactions([...transactions, { ...form, amount: parseFloat(form.amount), id: Date.now() }]);

                const d = new Date(form.date);

                setFilterYear(d.getFullYear());

                setFilterMonth(d.getMonth());

                setActiveTab(form.card);

                setForm({ ...form, name: '', amount: '' });

            };


            const deleteTransaction = (id) => {

                if(confirm("確定要刪除這筆紀錄嗎？")) setTransactions(transactions.filter(t => t.id !== id));

            };


            const startEdit = (t) => {

                setEditingId(t.id);

                setEditForm({ date: t.date, card: t.card, name: t.name, amount: t.amount });

            };


            const cancelEdit = () => {

                setEditingId(null);

                setEditForm({});

            };


            const saveEdit = () => {

                if (!editForm.amount || !editForm.name) { alert("請填寫完整的消費項目與金額"); return; }

                setTransactions(transactions.map(t => t.id === editingId ? { ...t, ...editForm, amount: parseFloat(editForm.amount) } : t));

                setEditingId(null);

                setEditForm({});

            };


            const changeMonth = (offset) => {

                let newMonth = filterMonth + offset;

                let newYear = filterYear;

                if (newMonth < 0) { newMonth = 11; newYear -= 1; }

                else if (newMonth > 11) { newMonth = 0; newYear += 1; }

                setFilterMonth(newMonth);

                setFilterYear(newYear);

            };


            const exportToExcel = () => {

                if (processedData.length === 0) { alert("目前沒有資料可以匯出。"); return; }

                const headers = ["日期", "信用卡", "品名", "消費金額", "帳單週期", "基本回饋率", "基本點數", "加碼回饋率", "加碼點數", "加碼入帳日", "超額不計入金額"];

                const rows = processedData.map(t => [t.date, CARD_CONFIG[t.card].name, t.name, t.amount, t.billingCycle, t.baseRateLabel, t.rewardBase, t.bonusRateLabel, t.rewardBonus, t.creditDateBonus, t.exceeded]);

                const csvContent = "\uFEFF" + [headers, ...rows].map(e => e.join(",")).join("\n");

                const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });

                const url = URL.createObjectURL(blob);

                const link = document.createElement("a");

                link.setAttribute("href", url);

                link.setAttribute("download", `星展回饋明細匯出_${formatDateLocal(new Date())}.csv`);

                document.body.appendChild(link);

                link.click();

                document.body.removeChild(link);

            };


            const importFromExcel = (e) => {

                const file = e.target.files[0];

                if (!file) return;

                const reader = new FileReader();

                reader.onload = (event) => {

                    const text = event.target.result;

                    const lines = text.split('\n').filter(l => l.trim() !== '');

                    if (lines.length <= 1) { alert('檔案無資料或格式不符'); return; }

                    const newTransactions = [];

                    for (let i = 1; i < lines.length; i++) {

                        const row = lines[i].split(',');

                        if (row.length < 4) continue;

                        let date = row[0].replace(/['"]/g, '').trim();

                        let cardStr = row[1].replace(/['"]/g, '').trim();

                        let name = row[2].replace(/['"]/g, '').trim();

                        let amount = parseFloat(row[3].replace(/['"]/g, '').trim());

                        if (isNaN(amount) || !date || !name) continue;


                        let cardKey = 'eco_linepay'; 

                        if (cardStr.includes('傳說')) cardKey = 'legend';

                        else if (cardStr.includes('海外') || cardStr.includes('4%')) cardKey = 'eco_overseas';

                        

                        newTransactions.push({ id: Date.now() + i + Math.random(), date, card: cardKey, name, amount });

                    }

                    if (newTransactions.length > 0) {

                        setTransactions(prev => [...prev, ...newTransactions]);

                        alert(`成功匯入 ${newTransactions.length} 筆資料！`);

                    } else {

                        alert('找不到有效的消費紀錄，請確認檔案格式是否為先前的匯出檔。');

                    }

                };

                reader.readAsText(file);

                if(fileInputRef.current) fileInputRef.current.value = '';

            };


            return (

                <div className="min-h-screen pb-24 font-sans bg-slate-50">

                    <input type="file" accept=".csv" ref={fileInputRef} onChange={importFromExcel} className="hidden" />


                    {/* AI Modal */}

                    {showAiModal && (

                        <div className="fixed inset-0 z-[100] flex items-center justify-center p-4 bg-black/60 backdrop-blur-sm">

                            <div className="bg-white w-full max-w-lg rounded-[2rem] shadow-2xl overflow-hidden">

                                <div className="bg-gradient-to-r from-red-600 to-pink-600 p-5 flex justify-between items-center text-white">

                                    <div className="flex items-center gap-2"><Icon name="Sparkles" size={20} /><h3 className="font-black">AI 智慧分析建議</h3></div>
