import React, { useState, useEffect, useMemo, useRef } from 'react';

import { 

  Plus, 

  Trash2, 

  Download, 

  Upload,

  CreditCard, 

  Calendar, 

  Monitor, 

  Smartphone, 

  AlertCircle,

  CheckCircle2,

  TrendingUp,

  History,

  ChevronLeft,

  ChevronRight,

  FileText,

  Target,

  Sparkles,

  Loader2,

  Pencil,

  X

} from 'lucide-react';


const App = () => {

  const [viewMode, setViewMode] = useState('pc'); 

  const [activeTab, setActiveTab] = useState('eco_linepay'); 

  

  const [aiAnalysis, setAiAnalysis] = useState("");

  const [isAiLoading, setIsAiLoading] = useState(false);

  const [showAiModal, setShowAiModal] = useState(false);


  const fileInputRef = useRef(null);


  const today = new Date();

  const [filterYear, setFilterYear] = useState(today.getFullYear());

  const [filterMonth, setFilterMonth] = useState(today.getMonth());


  // 載入資料並處理舊版 (v6) 資料的相容性

  const [transactions, setTransactions] = useState(() => {

    const saved = localStorage.getItem('dbs_transactions_v7');

    if (saved) return JSON.parse(saved);

    

    const savedV6 = localStorage.getItem('dbs_transactions_v6');

    if (savedV6) {

      const parsed = JSON.parse(savedV6);

      return parsed.map(t => {

        if (t.card === 'eco') return { ...t, card: 'eco_linepay' };

        return t;

      });

    }

    return [];

  });


  const [form, setForm] = useState({

    date: new Date().toISOString().split('T')[0],

    card: 'eco_linepay',

    name: '',

    amount: ''

  });


  // 編輯功能狀態

  const [editingId, setEditingId] = useState(null);

  const [editForm, setEditForm] = useState({});


  useEffect(() => {

    localStorage.setItem('dbs_transactions_v7', JSON.stringify(transactions));

  }, [transactions]);


  // 更新卡別設定：支援 eco卡 的兩種模式

  const CARD_CONFIG = {

    eco_linepay: { 

      name: 'eco 卡 (Line Pay 9%)', 

      shortName: 'eco (Line Pay)',

      baseRate: 0.01, 

      bonusRate: 0.09, 

      spendCap: 11111, 

      bonusPointCap: 1000 

    },

    eco_overseas: { 

      name: 'eco 卡 (海外 4%)', 

      shortName: 'eco (海外)',

      baseRate: 0.01, 

      bonusRate: 0.04, 

      spendCap: 15000, 

      bonusPointCap: 600

    },

    legend: { 

      name: '傳說對決卡', 

      shortName: '傳說卡',

      baseRate: 0.002, 

      bonusRate: 0.098, 

      spendCap: 11363, 

      bonusPointCap: 1000 

    }

  };


  // 卡別主題色系設定

  const getCardTheme = (cardKey) => {

    switch(cardKey) {

      case 'legend': 

        return { text: 'text-blue-700', badge: 'bg-blue-100 text-blue-700', border: 'border-l-4 border-l-blue-500' };

      case 'eco_overseas': 

        return { text: 'text-purple-700', badge: 'bg-purple-100 text-purple-700', border: 'border-l-4 border-l-purple-500' };

      default: // eco_linepay

        return { text: 'text-green-700', badge: 'bg-green-100 text-green-700', border: 'border-l-4 border-l-green-500' };

    }

  };


  const formatDateLocal = (date) => {

    const y = date.getFullYear();

    const m = String(date.getMonth() + 1).padStart(2, '0');

    const d = String(date.getDate()).padStart(2, '0');

    return `${y}-${m}-${d}`;

  };


  const callGeminiAPI = async (prompt) => {

    const apiKey = "";

    const url = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-09-2025:generateContent?key=${apiKey}`;

    

    const payload = {

      contents: [{ parts: [{ text: prompt }] }],

      systemInstruction: { 

        parts: [{ text: "你是一位專業的星展信用卡理財助手。請分析使用者的消費數據，給予簡短、專業且具備繁體中文口吻的理財建議。著重於如何最大化各卡別的加碼回饋。" }] 

      }

    };


    for (let i = 0; i < 5; i++) {

      try {

        const response = await fetch(url, {

          method: 'POST',

          headers: { 'Content-Type': 'application/json' },

          body: JSON.stringify(payload)

        });

        if (!response.ok) throw new Error('API Error');

        const data = await response.json();

        return data.candidates?.[0]?.content?.parts?.[0]?.text;

      } catch (error) {

        if (i === 4) throw error;

        await new Promise(r => setTimeout(r, Math.pow(2, i) * 1000));

      }

    }

  };


  const processedData = useMemo(() => {

    const sorted = [...transactions].sort((a, b) => new Date(a.date) - new Date(b.date));

    const monthlyUsage = {}; 


    return sorted.map((t) => {

      const dateObj = new Date(t.date);

      const day = dateObj.getDate();

      const amount = parseFloat(t.amount) || 0;

      const configKey = CARD_CONFIG[t.card] ? t.card : 'eco_linepay'; 

      const config = CARD_CONFIG[configKey];


      const consumeMonthKey = `${dateObj.getFullYear()}-${dateObj.getMonth()}_${configKey}`;

      const prevSpend = monthlyUsage[consumeMonthKey] || 0;

      

      const eligibleForBonus = Math.max(0, Math.min(amount, config.spendCap - prevSpend));

      const exceeded = Math.max(0, amount - eligibleForBonus);

      monthlyUsage[consumeMonthKey] = prevSpend + amount;


      let billYear = dateObj.getFullYear();

      let billMonth = dateObj.getMonth();

      if (day >= 12) {

        billMonth += 1;

        if (billMonth > 11) { billMonth = 0; billYear += 1; }

      }

      const billingCycleStr = `${billYear}/${String(billMonth + 1).padStart(2, '0')} 帳單`;


      const rewardBase = Math.round(amount * config.baseRate);

      const rewardBonus = Math.round(eligibleForBonus * config.bonusRate);

      

      const creditDateBonusObj = new Date(dateObj.getFullYear(), dateObj.getMonth() + 2, 1);

      

      return {

        ...t,

        card: configKey,

        eligibleForBonus,

        exceeded,

        rewardBase,

        rewardBonus,

        totalReward: rewardBase + rewardBonus,

        billingCycle: billingCycleStr,

        creditDateBonus: formatDateLocal(creditDateBonusObj),

        baseRateLabel: (config.baseRate * 100).toFixed(1) + '%',

        bonusRateLabel: (config.bonusRate * 100).toFixed(1) + '%',

        currentCap: config.spendCap

      };

    });

  }, [transactions]);


  const summary = useMemo(() => {

    const config = CARD_CONFIG[activeTab];

    const dataInMonth = processedData.filter(t => {

      const d = new Date(t.date);

      return d.getFullYear() === filterYear && d.getMonth() === filterMonth && t.card === activeTab;

    });


    const totalSpent = dataInMonth.reduce((sum, t) => sum + (parseFloat(t.amount) || 0), 0);

    const totalBonusPoints = dataInMonth.reduce((sum, t) => sum + t.rewardBonus, 0);

    const remainingQuota = Math.max(0, config.spendCap - totalSpent);

    const isReached = totalSpent >= config.spendCap;

    const isPointsMax = totalBonusPoints >= config.bonusPointCap;


    return {

      totalSpent,

      totalBonusPoints,

      remainingQuota,

      isReached,

      isPointsMax,

      percent: Math.min(100, (totalSpent / config.spendCap) * 100),

      dataInMonth,

      currentCap: config.spendCap,

      bonusPointCap: config.bonusPointCap

    };

  }, [processedData, activeTab, filterMonth, filterYear]);


  const handleAiAnalysis = async () => {

    setIsAiLoading(true);

    setShowAiModal(true);

    try {

      const spendDesc = summary.dataInMonth.map(t => `${t.name}($${t.amount})`).join(', ');

      const prompt = `

        當前分析卡別：${CARD_CONFIG[activeTab].name}

        查詢月份：${filterYear}年${filterMonth + 1}月

        目前已消費：$${summary.totalSpent}

        本卡別加碼上限：$${summary.currentCap}

        剩餘加碼額度：$${summary.remainingQuota}

        目前已領加碼點數：${summary.totalBonusPoints} 點 (上限 ${summary.bonusPointCap} 點)

        本月消費明細：${spendDesc || '無'}

        

        請根據以上資訊，給予這張卡片 3 點具體的理財策略建議，以極大化回饋。

      `;

      const result = await callGeminiAPI(prompt);

      setAiAnalysis(result);

    } catch (err) {

      setAiAnalysis("AI 分析暫時無法使用，請稍後再試。");

    } finally {

      setIsAiLoading(false);

    }

  };


  const addTransaction = (e) => {

    e.preventDefault();

    if (!form.amount || !form.name) return;

    setTransactions([...transactions, { ...form, amount: parseFloat(form.amount), id: Date.now() }]);

    

    const d = new Date(form.date);

    setFilterYear(d.getFullYear());

    setFilterMonth(d.getMonth());

    setActiveTab(form.card);

    setForm({ ...form, name: '', amount: '' });

  };


  const deleteTransaction = (id) => {

    if(confirm("確定要刪除這筆紀錄嗎？")) {

        setTransactions(transactions.filter(t => t.id !== id));

    }

  };


  // 編輯相關操作

  const startEdit = (t) => {

    setEditingId(t.id);

    setEditForm({ date: t.date, card: t.card, name: t.name, amount: t.amount });

  };


  const cancelEdit = () => {

    setEditingId(null);

    setEditForm({});

  };


  const saveEdit = () => {

    if (!editForm.amount || !editForm.name) {

      alert("請填寫完整的消費項目與金額");

      return;

    }

    setTransactions(transactions.map(t => 

      t.id === editingId ? { ...t, ...editForm, amount: parseFloat(editForm.amount) } : t

    ));

    setEditingId(null);

    setEditForm({});

  };


  const changeMonth = (offset) => {

    let newMonth = filterMonth + offset;

    let newYear = filterYear;

    if (newMonth < 0) { newMonth = 11; newYear -= 1; }

    else if (newMonth > 11) { newMonth = 0; newYear += 1; }

    setFilterMonth(newMonth);

    setFilterYear(newYear);

  };


  // 匯出 CSV

  const exportToExcel = () => {

    if (processedData.length === 0) {

      alert("目前沒有資料可以匯出。");

      return;

    }

    const headers = ["日期", "信用卡", "品名", "消費金額", "帳單週期", "基本回饋率", "基本點數", "加碼回饋率", "加碼點數", "加碼入帳日", "超額不計入金額"];

    const rows = processedData.map(t => [

      t.date, CARD_CONFIG[t.card].name, t.name, t.amount, t.billingCycle, t.baseRateLabel, t.rewardBase, t.bonusRateLabel, t.rewardBonus, t.creditDateBonus, t.exceeded

    ]);

    const csvContent = "\uFEFF" + [headers, ...rows].map(e => e.join(",")).join("\n");

    const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });

    const url = URL.createObjectURL(blob);

    const link = document.createElement("a");

    link.setAttribute("href", url);

    link.setAttribute("download", `星展回饋明細匯出_${formatDateLocal(new Date())}.csv`);

    document.body.appendChild(link);

    link.click();

    document.body.removeChild(link);

  };


  // 匯入 CSV

  const importFromExcel = (e) => {

    const file = e.target.files[0];

    if (!file) return;


    const reader = new FileReader();

    reader.onload = (event) => {

      const text = event.target.result;

      const lines = text.split('\n').filter(l => l.trim() !== '');

      if (lines.length <= 1) {

        alert('檔案無資料或格式不符');

        return;

      }


      const newTransactions = [];

      // 跳過表頭列，從 index 1 開始

      for (let i = 1; i < lines.length; i++) {

        // 簡易分割處理

        const row = lines[i].split(',');

        if (row.length < 4) continue;


        let date = row[0].replace(/['"]/g, '').trim();

        let cardStr = row[1].replace(/['"]/g, '').trim();

        let name = row[2].replace(/['"]/g, '').trim();

        let amount = parseFloat(row[3].replace(/['"]/g, '').trim());


        if (isNaN(amount) || !date || !name) continue;


        // 智慧判斷卡別

        let cardKey = 'eco_linepay'; // 預設

        if (cardStr.includes('傳說')) {

          cardKey = 'legend';

        } else if (cardStr.includes('海外') || cardStr.includes('4%')) {

          cardKey = 'eco_overseas';

        } else if (cardStr.includes('eco') || cardStr.includes('Line Pay') || cardStr.includes('9%')) {

          cardKey = 'eco_linepay';

        }


        newTransactions.push({

          id: Date.now() + i + Math.random(),

          date,

          card: cardKey,

          name,

          amount

        });

      }


      if (newTransactions.length > 0) {

        setTransactions(prev => [...prev, ...newTransactions]);

        alert(`成功匯入 ${newTransactions.length} 筆資料！`);

      } else {

        alert('找不到有效的消費紀錄，請確認檔案格式是否為先前的匯出檔。');

      }

    };

    reader.readAsText(file);

    // 重設 input

    if(fileInputRef.current) fileInputRef.current.value = '';

  };


  return (

    <div className="min-h-screen bg-slate-50 text-slate-900 font-sans pb-24">

      {/* 隱藏的檔案上傳元件 */}

      <input 

        type="file" 

        accept=".csv" 

        ref={fileInputRef} 

        onChange={importFromExcel} 

        className="hidden" 

      />


      {/* AI 分析 Modal */}

      {showAiModal && (

        <div className="fixed inset-0 z-[100] flex items-center justify-center p-4 bg-black/60 backdrop-blur-sm">

          <div className="bg-white w-full max-w-lg rounded-[2.5rem] shadow-2xl overflow-hidden animate-in fade-in zoom-in duration-300">

            <div className="bg-gradient-to-r from-red-600 to-pink-600 p-6 flex justify-between items-center text-white">

              <div className="flex items-center gap-2">

                <Sparkles size={24} />

                <h3 className="font-black text-lg">AI 智慧分析建議</h3>

              </div>

              <button onClick={() => setShowAiModal(false)} className="bg-white/20 hover:bg-white/30 p-2 rounded-full transition">

                <X size={18} />

              </button>

            </div>

            <div className="p-8">

              {isAiLoading ? (

                <div className="flex flex-col items-center justify-center py-12 space-y-4">

                  <Loader2 size={48} className="text-red-500 animate-spin" />

                  <p className="text-slate-400 font-bold animate-pulse">理財助手正在分析您的消費習慣...</p>

                </div>

              ) : (

                <div className="prose prose-slate max-h-[60vh] overflow-y-auto pr-2 custom-scrollbar">

                  <div className="text-slate-700 leading-relaxed whitespace-pre-wrap font-medium">

                    {aiAnalysis}

                  </div>

                  <div className="mt-8 p-4 bg-slate-50 rounded-2xl border border-slate-100 flex items-start gap-3">

                    <Target size={18} className="text-red-500 mt-1 shrink-0" />

                    <p className="text-[11px] text-slate-400">

                        AI 建議基於您輸入的數據計算得出。信用卡權益可能隨銀行公告變更，實際回饋請以官方為準。

                    </p>

                  </div>

                </div>

              )}

            </div>

          </div>

        </div>

      )}


      {/* Header */}

      <nav className="bg-red-600 text-white p-4 sticky top-0 z-50 shadow-md">

        <div className="max-w-7xl mx-auto flex justify-between items-center">

          <div className="flex items-center gap-2">

            <CreditCard size={24} />

            <h1 className="text-xl font-bold tracking-tight">星展雙卡回饋計算器 <span className="text-xs font-normal opacity-70 italic">v2026.4</span></h1>

          </div>

          <div className="flex bg-red-700 rounded-lg p-1">

            <button onClick={() => setViewMode('pc')} className={`p-1.5 rounded ${viewMode === 'pc' ? 'bg-white text-red-600' : 'text-white'}`}><Monitor size={18}/></button>

            <button onClick={() => setViewMode('mobile')} className={`p-1.5 rounded ${viewMode === 'mobile' ? 'bg-white text-red-600' : 'text-white'}`}><Smartphone size={18}/></button>

          </div>

        </div>

      </nav>


      <main className="max-w-7xl mx-auto p-4 space-y-6">

        {/* Dashboard Section */}

        <section className="grid grid-cols-1 lg:grid-cols-3 gap-4">

          <div className="lg:col-span-2 bg-white p-6 rounded-3xl shadow-sm border border-slate-200">

            <div className="flex flex-col xl:flex-row justify-between items-start xl:items-end mb-6 gap-4">

              <div>

                <div className="flex items-center gap-2 text-slate-500 mb-3">

                  <button onClick={() => changeMonth(-1)} className="p-1 hover:bg-slate-100 rounded-full transition"><ChevronLeft size={18}/></button>

                  <h2 className="text-lg font-black text-slate-800">{filterYear}年 {filterMonth + 1}月 消費統計</h2>

                  <button onClick={() => changeMonth(1)} className="p-1 hover:bg-slate-100 rounded-full transition"><ChevronRight size={18}/></button>

                </div>

                {/* Tabs */}

                <div className="flex flex-wrap gap-2">

                  {Object.keys(CARD_CONFIG).map(cardKey => (

                    <button 

                      key={cardKey}

                      onClick={() => setActiveTab(cardKey)}

                      className={`px-4 py-2 rounded-full text-xs font-bold transition shadow-sm ${activeTab === cardKey ? 'bg-red-600 text-white' : 'bg-slate-100 text-slate-500 hover:bg-slate-200'}`}

                    >

                      {CARD_CONFIG[cardKey].shortName}

                    </button>

                  ))}

                </div>

              </div>

              <div className="text-right flex flex-col items-start xl:items-end w-full xl:w-auto">

                <div className="flex items-center gap-2 mb-1 w-full justify-between xl:justify-end">

                    <span className="text-xs font-bold text-slate-400 block xl:hidden">{CARD_CONFIG[activeTab].name}</span>

                    <button 

                        onClick={handleAiAnalysis}

                        className="flex items-center gap-1.5 bg-gradient-to-r from-red-500 to-pink-500 text-white text-[10px] font-black px-3 py-1 rounded-full shadow-lg shadow-red-200 hover:scale-105 transition-transform active:scale-95"

                    >

                        <Sparkles size={12} /> AI 智慧分析 ✨

                    </button>

                </div>

                <div className="mt-2 xl:mt-0">

                    <span className={`text-3xl font-black ${summary.isReached ? 'text-red-600' : 'text-slate-800'}`}>${summary.totalSpent.toLocaleString()}</span>

                    <span className="text-slate-400 text-sm font-medium ml-1"> / ${summary.currentCap.toLocaleString()}</span>

                </div>

              </div>

            </div>

            

            <div className="w-full bg-slate-100 h-5 rounded-full overflow-hidden mb-5 border border-slate-100 shadow-inner">

              <div className={`h-full transition-all duration-1000 ease-out ${summary.isReached ? 'bg-red-500' : 'bg-green-500'}`} style={{ width: `${summary.percent}%` }}></div>

            </div>


            <div className="grid grid-cols-1 md:grid-cols-2 gap-4">

              <div className="flex items-center gap-3 bg-blue-50 p-3 rounded-2xl">

                <div className="bg-blue-600 text-white p-2 rounded-xl"><TrendingUp size={20}/></div>

                <div>

                  <p className="text-[10px] text-blue-600 font-bold uppercase">加碼餘額</p>

                  <p className="text-sm font-black text-slate-800">還剩 <span className="text-blue-600">${summary.remainingQuota.toLocaleString()}</span> 可刷</p>

                </div>
